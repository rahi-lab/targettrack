# Step 1: Clone the repository
git clone https://github.com/lpbsscientist/targettrack.git
cd targettrack/

# Step 2: Create and activate the conda environment with matching Python version
conda create -n tt_py38 python=3.8.15 -y
conda activate tt_py38

# Step 3: Install PyTorch
pip install torch==1.7.1 torchvision==0.8.2 torchaudio

# Step 4: Install core Python packages
conda install ipython=7.19.0 matplotlib=3.3.2 pandas=1.1.3 scikit-image=0.18.3 -y
pip install numpy==1.24.4 scikit-learn==1.3.2 scipy==1.10.1 tqdm==4.50.2

# Step 5: Install required pip-only packages
pip install \
    umap-learn==0.5.3 \
    sparse==0.13.0 \
    nd2reader==3.2.3 \
    PyQt5==5.12 \
    pyqtgraph==0.11.1 \
    opencv-python==4.5.1.48 \
    opencv-python-headless==4.4.0.46 \
    h5py==3.1.0 \
    albumentations==0.5.2 \
    connected-components-3d==2.0.0 \
    alphashape==1.3.1 \
    slicerator==1.1.0 \
    pims==0.7 \
    imgaug==0.4.0 \
    click-log==0.4.0 \
    trimesh==4.6.8 \
    xmltodict==0.14.2 \
    rtree==1.3.0

# Step 6: Compile and install the GMMREG C++ extension
cd gmmreg-python/src
pip install .
cd ../..
